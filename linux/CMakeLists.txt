find_package(Qt6 6.4 REQUIRED COMPONENTS Widgets)
find_package(SDL3 3.2 REQUIRED CONFIG)
add_library(bf2_setup_linux STATIC
  ../src/window_qt.cpp ../src/client_qt.cpp ../src/microphone_sdl.cpp)
target_link_libraries(bf2_setup_linux PUBLIC bf2_setup_session bf2_setup_boost Qt6::Widgets SDL3::SDL3)
target_compile_options(bf2_setup_linux PRIVATE -Wall -Wextra -Wpedantic)
add_executable(OpenBF2 ../src/main_qt.cpp ../src/setup.cpp ../src/installer.cpp ../src/store.cpp)
target_link_libraries(OpenBF2 PRIVATE bf2_setup_linux)
set_target_properties(OpenBF2 PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin" INSTALL_RPATH "$ORIGIN/../lib")
