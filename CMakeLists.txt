cmake_minimum_required(VERSION 3.24)
project(OpenBF2Setup VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(APPLE)
  set(CMAKE_OSX_DEPLOYMENT_TARGET 11.0 CACHE STRING "Minimum macOS")
  if(NOT CMAKE_OSX_ARCHITECTURES)
    set(CMAKE_OSX_ARCHITECTURES "arm64;x86_64" CACHE STRING "Universal architectures" FORCE)
  endif()
endif()
find_package(ZLIB REQUIRED)
find_package(Threads REQUIRED)
find_program(SETUP_CARGO cargo HINTS "$ENV{HOME}/.cargo/bin" REQUIRED)
set(SETUP_RTPTOOL "${CMAKE_SOURCE_DIR}/third_party/rtptool" CACHE PATH "Owner rtptool bf2-0x0320 source; submodule-ready snapshot")
# Cargo's path is relative to its manifest. An alternate owner checkout is selected
# through a build-local manifest, leaving both source repositories unchanged.
file(MAKE_DIRECTORY "${CMAKE_BINARY_DIR}/rtp-bridge")
configure_file(core/rtp-bridge/Cargo.lock rtp-bridge/Cargo.lock COPYONLY)
file(READ "${CMAKE_SOURCE_DIR}/core/rtp-bridge/Cargo.toml" bridge_manifest)
string(REPLACE "../../third_party/rtptool" "${SETUP_RTPTOOL}" bridge_manifest "${bridge_manifest}")
string(REPLACE "crate-type = [\"staticlib\"]" "crate-type = [\"staticlib\"]\npath = \"${CMAKE_SOURCE_DIR}/core/rtp-bridge/src/lib.rs\"" bridge_manifest "${bridge_manifest}")
file(WRITE "${CMAKE_BINARY_DIR}/rtp-bridge/Cargo.toml" "${bridge_manifest}")
file(GLOB_RECURSE rust_sources CONFIGURE_DEPENDS "${CMAKE_SOURCE_DIR}/core/rtp-bridge/src/*.rs" "${SETUP_RTPTOOL}/src/*.rs")
if(APPLE)
  set(rust_targets)
  foreach(arch IN LISTS CMAKE_OSX_ARCHITECTURES)
    if(arch STREQUAL "arm64")
      list(APPEND rust_targets aarch64-apple-darwin)
    elseif(arch STREQUAL "x86_64")
      list(APPEND rust_targets x86_64-apple-darwin)
    else()
      message(FATAL_ERROR "Unsupported Apple architecture: ${arch}")
    endif()
  endforeach()
else()
  execute_process(COMMAND "${SETUP_CARGO}" -vV OUTPUT_VARIABLE cargo_version COMMAND_ERROR_IS_FATAL ANY)
  string(REGEX MATCH "host: ([^\n]+)" host "${cargo_version}")
  set(rust_targets "${CMAKE_MATCH_1}")
endif()
set(rust_libraries)
foreach(triple IN LISTS rust_targets)
  if(WIN32)
    set(lib "${CMAKE_BINARY_DIR}/cargo/${triple}/release/openbf2_rtp_bridge.lib")
  else()
    set(lib "${CMAKE_BINARY_DIR}/cargo/${triple}/release/libopenbf2_rtp_bridge.a")
  endif()
  add_custom_command(OUTPUT "${lib}"
    COMMAND "${SETUP_CARGO}" build --release --offline --locked --manifest-path "${CMAKE_BINARY_DIR}/rtp-bridge/Cargo.toml" --target "${triple}" --target-dir "${CMAKE_BINARY_DIR}/cargo"
    DEPENDS ${rust_sources} core/rtp-bridge/Cargo.toml core/rtp-bridge/Cargo.lock "${SETUP_RTPTOOL}/Cargo.toml" VERBATIM)
  list(APPEND rust_libraries "${lib}")
endforeach()
if(APPLE)
  set(rust_library "${CMAKE_BINARY_DIR}/libopenbf2_rtp.a")
  add_custom_command(OUTPUT "${rust_library}" COMMAND /usr/bin/lipo -create ${rust_libraries} -output "${rust_library}" DEPENDS ${rust_libraries} VERBATIM)
else()
  set(rust_library ${rust_libraries})
endif()
add_custom_target(bf2_setup_rtp DEPENDS "${rust_library}")
add_library(bf2_setup_core STATIC src/media.cpp src/cabinet.cpp src/edition.cpp src/patch.cpp src/install.cpp src/digest.cpp src/key_check.cpp)
add_dependencies(bf2_setup_core bf2_setup_rtp)
target_include_directories(bf2_setup_core PUBLIC src)
target_link_libraries(bf2_setup_core PUBLIC "${rust_library}" ZLIB::ZLIB Threads::Threads ${CMAKE_DL_LIBS})
set(SETUP_SCRATCH "${CMAKE_BINARY_DIR}/scratch" CACHE PATH "Developer scratch output root")
file(MAKE_DIRECTORY "${SETUP_SCRATCH}")
target_compile_definitions(bf2_setup_core PRIVATE OPENBF2_SETUP_SCRATCH="${SETUP_SCRATCH}")
if(MSVC)
  target_compile_options(bf2_setup_core PRIVATE /W4)
  target_link_libraries(bf2_setup_core PUBLIC userenv ws2_32 bcrypt ntdll)
else()
  target_compile_options(bf2_setup_core PRIVATE -Wall -Wextra -Wpedantic)
endif()
if(APPLE)
  target_link_libraries(bf2_setup_core PUBLIC "-framework Security" "-framework CoreFoundation" resolv)
elseif(UNIX)
  target_link_libraries(bf2_setup_core PUBLIC m rt util)
endif()
if(WIN32)
  add_library(bf2_setup_platform STATIC src/windows.cpp)
else()
  add_library(bf2_setup_platform STATIC src/posix.cpp)
endif()
target_include_directories(bf2_setup_platform PUBLIC src)
add_executable(bf2_setup cli/main.cpp)
target_link_libraries(bf2_setup PRIVATE bf2_setup_core bf2_setup_platform)
include(CTest)
if(BUILD_TESTING)
  add_executable(bf2_setup_key_checks tests/key_check.cpp)
  target_link_libraries(bf2_setup_key_checks PRIVATE bf2_setup_core)
  add_test(NAME key_checks COMMAND bf2_setup_key_checks)
  add_test(NAME key_checks_owner COMMAND bf2_setup_key_checks --owner-keys)
  set_tests_properties(key_checks_owner PROPERTIES SKIP_RETURN_CODE 77)
  add_executable(bf2_setup_checks tests/macos_setup.cpp)
  target_link_libraries(bf2_setup_checks PRIVATE bf2_setup_core bf2_setup_platform)
  target_compile_definitions(bf2_setup_checks PRIVATE SETUP_TEST_ROOT="${SETUP_SCRATCH}")
  add_executable(bf2_setup_transaction_checks tests/transaction_local.cpp)
  target_link_libraries(bf2_setup_transaction_checks PRIVATE bf2_setup_core bf2_setup_platform)
  target_compile_definitions(bf2_setup_transaction_checks PRIVATE SETUP_TEST_ROOT="${SETUP_SCRATCH}")
  add_test(NAME setup_checks COMMAND bf2_setup_checks "${SETUP_SCRATCH}/fixtures")
endif()
option(SETUP_BUILD_APP "Build native macOS setup assistant" ON)
if(APPLE AND SETUP_BUILD_APP AND EXISTS "${CMAKE_SOURCE_DIR}/macos/CMakeLists.txt")
  add_subdirectory(macos)
endif()
